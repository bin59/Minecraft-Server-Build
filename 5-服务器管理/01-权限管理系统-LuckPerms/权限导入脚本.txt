############################################################
#  LuckPerms 权限一键导入脚本
#  服务器：南瓜生存服 | Leaf 1.21.11 | MC 1.21.11
#  版本：v2（2026-09-26，身份精简：default / admin / owner + bedrock）
#
#  【使用方法】
#  1. 确认 plugins/ 下已有 LuckPerms-Bukkit-*.jar 主插件并已启动过一次
#  2. 将 plugins/LuckPerms/config.yml 替换为配套的 config.yml
#  3. 在【服务端控制台】逐段粘贴执行（不要在游戏内聊天框粘贴，会超长被截断）
#  4. 全部执行完后执行 /lp reload
#
#  【幂等性】
#  本脚本全部使用 creategroup / setweight / setprefix / parent add / permission set，
#  重复执行不会产生重复条目，也不会报错。
#
#  【说明】
#  - "permission set <节点>" 省略 true 即表示 true，下文统一省略
#  - 带 "# 可选" 的行默认不执行，按需去掉注释后单独执行
#  - 标注 ⚠️ 的节点请先在测试服验证（不同插件版本节点名可能有差异）
############################################################


############################################################
#  第 1 段：创建权限组 + 权重 + 前缀
############################################################

lp creategroup default
lp creategroup admin
lp creategroup owner
lp creategroup bedrock

lp group default setweight 1
lp group admin   setweight 80
lp group owner   setweight 100
lp group bedrock setweight 0

lp group default meta setprefix 1 "&8[&7新手&8]&r"
lp group admin   meta setprefix 1 "&8[&c管理&8]&r"
lp group owner   meta setprefix 1 "&8[&4服主&8]&r"


############################################################
#  第 2 段：继承链
#    default <- admin <- owner
#    bedrock 为独立附加组，不参与继承
############################################################

lp group admin parent add default
lp group owner parent add admin
